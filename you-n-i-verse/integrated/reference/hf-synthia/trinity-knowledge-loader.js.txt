// Initialize system
async function initializeSystem() {
    updateStatus('Checking your purchased frameworks...');
    
    // Load frameworks user owns
    await loadPurchasedFrameworks();
    
    // Display marketplace for unpurchased frameworks
    displayMarketplace();
    
    // Enable search
    document.getElementById('searchInput').disabled = false;
    document.getElementById('searchBtn').disabled = false;
    document.getElementById('searchInput').focus();
    
    const count = window.LOADED_FRAMEWORKS.size;
    updateStatus(`Ready! ${count} framework${count !== 1 ? 's' : ''} loaded`);
}

async function loadPurchasedFrameworks() {
    for (const frameworkKey of USER_PURCHASES.owned) {
        await loadFramework(frameworkKey);
    }
}

async function loadFramework(frameworkKey) {
    const framework = FRAMEWORK_CATALOG[frameworkKey];
    if (!framework) return;

    updateLoadingBadge(framework.name, true);

    try {
        // Load the knowledge module
        await loadScript(`./modules/${framework.file}`);
        
        window.LOADED_FRAMEWORKS.add(frameworkKey);
        updateLoadingBadge(framework.name, false);
        
        console.log(`✓ Loaded ${framework.name}`);
    } catch (error) {
        console.error(`Failed to load ${framework.name}:`, error);
        updateLoadingBadge(framework.name, false, true);
    }
}

function loadScript(src) {
    return new Promise((resolve, reject) => {
        const script = document.createElement('script');
        script.src = src;
        script.onload = resolve;
        script.onerror = () => {
            // If file doesn't exist, load demo data instead
            console.warn(`Could not load ${src}, using demo data`);
            loadDemoData(src);
            resolve();
        };
        document.head.appendChild(script);
    });
}

function loadDemoData(src) {
    // For demo purposes, load inline data
    if (src.includes('trinity-engine')) {
        loadTrinityEngineDemo();
    } else if (src.includes('sentence-generator')) {
        loadSentenceGeneratorDemo();
    }
}

function updateStatus(message) {
    document.getElementById('knowledgeStatus').textContent = message;
}

function updateLoadingBadge(name, loading, error = false) {
    const container = document.getElementById('loadedFrameworks');
    const badgeId = `badge-${name.replace(/\s/g, '-')}`;
    
    if (loading) {
        const badge = document.createElement('div');
        badge.className = 'framework-badge loading';
        badge.id = badgeId;
        badge.textContent = `Loading ${name}...`;
        container.appendChild(badge);
    } else {
        const badge = document.getElementById(badgeId);
        if (badge) {
            badge.className = 'framework-badge';
            badge.textContent = error ? `❌ ${name}` : `✓ ${name}`;
        }
    }
}

// Search functionality
function searchKnowledge() {
    const query = document.getElementById('searchInput').value.toLowerCase().trim();
    if (!query) return;

    const results = [];

    // Search through all loaded frameworks
    for (const [frameworkKey, framework] of Object.entries(window.TRINITY_KNOWLEDGE)) {
        for (const chunk of framework.chunks) {
            const score = calculateRelevance(query, chunk.keywords, chunk.answer);
            if (score > 0.2) {
                results.push({
                    framework: framework.name,
                    chunk: chunk,
                    score: score
                });
            }
        }
    }

    results.sort((a, b) => b.score - a.score);
    displayResults(query, results);
}

function calculateRelevance(query, keywords, answerText = "") {
    let score = 0;
    const queryWords = query.split(/\s+/);

    // Exact keyword matches
    for (const keyword of keywords) {
        if (query.includes(keyword.toLowerCase())) {
            score += 0.5;
        }
    }

    // Word overlap
    for (const word of queryWords) {
        if (word.length < 3) continue;
        
        for (const keyword of keywords) {
            if (keyword.toLowerCase().includes(word)) {
                score += 0.2;
            }
        }

        if (answerText && answerText.toLowerCase().includes(word)) {
            score += 0.1;
        }
    }

    return score;
}

function displayResults(query, results) {
    const container = document.getElementById('answerContainer');
    container.innerHTML = '';
    container.classList.add('active');

    if (results.length === 0) {
        container.innerHTML = `
            <div class="answer-card">
                <div class="no-results">
                    <h3 style="color: #a78bfa; margin-bottom: 1rem;">No results found</h3>
                    <p style="margin-bottom: 1rem;">
                        Try rephrasing your question or unlock more frameworks for deeper knowledge.
                    </p>
                    ${window.LOADED_FRAMEWORKS.size < Object.keys(FRAMEWORK_CATALOG).length ? 
                        '<p style="color: #ec4899;">💎 More frameworks available below</p>' : ''}
                </div>
            </div>
        `;
        return;
    }

    // Show top 3 results
    results.slice(0, 3).forEach(result => {
        const card = document.createElement('div');
        card.className = 'answer-card';
        card.innerHTML = `
            <div class="answer-header">
                <div class="question-echo">${result.chunk.question}</div>
                <div class="relevance-score">${Math.round(result.score * 100)}% match</div>
            </div>
            <div class="answer-content">
                ${formatAnswer(result.chunk.answer)}
            </div>
            <div class="source-section">
                <div class="source-label">Source:</div>
                <div class="source-info">
                    <span class="source-badge">${result.framework}</span>
                    <span>${result.chunk.category}</span>
                </div>
            </div>
        `;
        container.appendChild(card);
    });
}

function formatAnswer(text) {
    return text
        .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
        .split('\n')
        .map(line => line.trim() ? `<p>${line}</p>` : '')
        .join('');
}

function askExample(question) {
    document.getElementById('searchInput').value = question;
    searchKnowledge();
}

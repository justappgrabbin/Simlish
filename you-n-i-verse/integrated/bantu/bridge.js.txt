const nato=['Alfa','Bravo','Charlie','Delta','Echo','Foxtrot','Golf','Hotel','India','Juliett','Kilo','Lima','Mike','November','Oscar','Papa','Quebec','Romeo','Sierra','Tango','Uniform','Victor','Whiskey','X-ray','Yankee','Zulu'];
// Escape 63 followed by two 6-bit chunks for each UTF-8 byte >=63; 63 itself escaped.
// The first 6-bit chunk stores the high two bits, second the low six bits.
function encodeBytes(bytes){let out=[];for(const b of bytes){if(b<63)out.push(b);else out.push(63,b>>6,b&63)}return out}
function decodeBytes(codes){let out=[];for(let i=0;i<codes.length;i++){let v=codes[i];if(v===63){if(i+2>=codes.length)throw Error('Truncated escape');const hi=codes[++i],lo=codes[++i];if(hi>3||lo>63)throw Error('Invalid escape');v=(hi<<6)|lo;if(v<63)throw Error('Noncanonical escape')}if(v>255||v<0)throw Error('Invalid byte');out.push(v)}return Uint8Array.from(out)}
function bits(v){return v.toString(2).padStart(6,'0')}
function encodeText(text){const bytes=new TextEncoder().encode(text),codes=encodeBytes(bytes);return {bytes,codes,groups:codes.map((v,i)=>({id:i,value:v,bits:bits(v),levels:[bits(v).slice(0,2),bits(v).slice(2,4),bits(v).slice(4,6)]}))}}
function decodeText(codes){return new TextDecoder('utf-8',{fatal:true}).decode(decodeBytes(codes))}
function representations(text){const chars=Array.from(text),enc=encodeText(text);return {characters:chars.length,utf8:Array.from(enc.bytes,b=>b.toString(2).padStart(8,'0')).join(' '),ascii:chars.map(c=>c.codePointAt(0)<128?c.codePointAt(0):'non-ASCII').join(' '),nato:chars.map(c=>/^[A-Za-z]$/.test(c)?nato[c.toUpperCase().charCodeAt(0)-65]:c===' '?'[space]':'[unmapped]').join(' '),unicode:chars.map(c=>'U+'+c.codePointAt(0).toString(16).toUpperCase().padStart(4,'0')).join(' '),hexagrams:enc.groups.map(g=>g.bits).join(' '),decoded:decodeText(enc.codes),roundTrip:decodeText(enc.codes)===text}}
const cvs=document.getElementById('field'),ctx=cvs.getContext('2d');let particles=[],groups=[],running=true,frame=0,last=0;
function setup(){const s=document.getElementById('input').value;const enc=encodeText(s);groups=enc.groups;particles=[];const cols=Math.max(1,Math.ceil(Math.sqrt(groups.length*2)));for(let i=0;i<groups.length;i++){const g=groups[i],cx=55+(i%cols)*(790/cols),cy=55+Math.floor(i/cols)*65;for(let j=0;j<6;j++){const angle=j*Math.PI/3;particles.push({group:i,bit:Number(g.bits[j]),level:Math.floor(j/2),x:cx+Math.cos(angle)*17,y:cy+Math.sin(angle)*17,vx:0,vy:0,homeX:cx+Math.cos(angle)*17,homeY:cy+Math.sin(angle)*17})}}const r=representations(s);document.getElementById('out').textContent='UTF-8 binary: '+r.utf8+'\nASCII decimal: '+r.ascii+'\nNATO: '+r.nato+'\nUnicode: '+r.unicode+'\n6-bit hexagrams: '+r.hexagrams+'\nRound-trip decoded: '+r.decoded+'\nVerified exact text: '+r.roundTrip;frame=0;draw()}
function tick(){const strength=Number(document.getElementById('speed').value),mode=document.getElementById('mode').value;for(let i=0;i<particles.length;i++){const p=particles[i];let fx=(p.homeX-p.x)*0.005,fy=(p.homeY-p.y)*0.005;for(let j=0;j<particles.length;j++){if(i===j)continue;const q=particles[j],dx=q.x-p.x,dy=q.y-p.y,d2=dx*dx+dy*dy+24;if(d2>16000)continue;const d=Math.sqrt(d2),same=p.group===q.group;let f=(same?0.18:0.035)*strength/(1+d2/500);if(mode==='repel')f=-f;if(mode==='orbit'){fx+=-dy/d*f;fy+=dx/d*f}else{fx+=dx/d*f;fy+=dy/d*f}if(d<9){fx-=dx/d*0.13;fy-=dy/d*0.13}}p.vx=(p.vx+fx)*0.88;p.vy=(p.vy+fy)*0.88}for(const p of particles){p.x=Math.max(5,Math.min(895,p.x+p.vx));p.y=Math.max(5,Math.min(425,p.y+p.vy))}frame++;draw()}
function draw(){ctx.clearRect(0,0,900,430);const colors=['#b48aff','#58dec9','#f6c36b'];for(let i=0;i<particles.length;i+=6){ctx.beginPath();for(let j=0;j<6;j++){const p=particles[i+j],q=particles[i+(j+1)%6];ctx.moveTo(p.x,p.y);ctx.lineTo(q.x,q.y)}ctx.strokeStyle='#68557f';ctx.lineWidth=1;ctx.stroke()}for(const p of particles){ctx.beginPath();ctx.arc(p.x,p.y,p.bit?5:3,0,Math.PI*2);ctx.fillStyle=colors[p.level];ctx.globalAlpha=p.bit?1:.45;ctx.fill();ctx.globalAlpha=1}document.getElementById('stats').textContent=`${groups.length} six-bit structures · ${particles.length} primitive positions · ${frame} simulation steps · ${running?'running':'paused'}`}
function loop(t){if(t-last>32){last=t;if(running)tick()}requestAnimationFrame(loop)}document.getElementById('build').onclick=setup;document.getElementById('reset').onclick=setup;document.getElementById('pause').onclick=()=>{running=!running;document.getElementById('pause').textContent=running?'Pause':'Resume';draw()};document.getElementById('step').onclick=()=>{running=false;document.getElementById('pause').textContent='Resume';tick()};document.getElementById('speed').oninput=e=>document.getElementById('strength').textContent=e.target.value;document.getElementById('mode').onchange=draw;
window.BANTU={encodeText,decodeText,representations};setup();requestAnimationFrame(loop);
